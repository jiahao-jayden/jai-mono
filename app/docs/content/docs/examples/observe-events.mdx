title: 示例：观测事件流
description: 完整订阅流式文本渲染、思考过程、工具调用与上下文压缩事件。
---

本示例展示如何在终端中完整接入 Agent 的事件总线，渲染实时的流式打字效果与工具进度条。

```ts
import { createCodingAgent } from "@jai/coding-agent";

const created = await createCodingAgent({
  model: "anthropic/claude-3-7-sonnet",
  cwd: process.cwd(),
});

if (created.isErr()) {
  console.error("创建失败:", created.error.message);
  process.exit(1);
}

const agent = created.value;

// 1. 订阅事件流
const unsubscribe = agent.subscribe((event) => {
  switch (event.type) {
    case "agent_start":
      console.log("▶ [任务启动]");
      break;

    case "message_update": {
      const ae = event.assistantEvent;
      if (ae.type === "text_delta") {
        // 模型输出常规文本 Token
        process.stdout.write(ae.delta);
      } else if (ae.type === "thinking_delta") {
        // 模型输出思考过程（Reasoning）Token，可折叠或灰色打印
        process.stdout.write(`\x1b[90m${ae.delta}\x1b[0m`);
      } else if (ae.type === "toolcall_end") {
        // 参数组装就绪
        console.log(`\n⚙ [模型请求调用工具] ${ae.toolCall.name}`);
      }
      break;
    }

    case "tool_execution_start":
      console.log(`\n⏳ [工具开始执行] ${event.toolName} -> ${event.title}`);
      break;

    case "tool_execution_end":
      const statusIcon = event.isError ? "❌" : "✅";
      console.log(`${statusIcon} [工具执行完毕] ${event.toolName}`);
      break;

    case "compaction_start":
      console.log(`\n✂ [上下文自动压缩] 触发原因: ${event.trigger}`);
      break;

    case "background_agent_end":
      console.log(`\n🏁 [后台任务完成] ${event.title} (状态: ${event.status})`);
      break;

    case "agent_end":
      console.log("\n■ [整轮执行结束]");
      break;
  }
});

// 2. 发起任务并等待空闲
await agent.prompt("分析当前目录下的 git 状态，如果有改动，给出精炼的修改摘要");
await agent.waitForIdle();

// 3. 注销监听与安全退出
unsubscribe();
await agent.close();
```
