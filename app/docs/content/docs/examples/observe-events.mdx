# 示例：观测事件流

```ts
import { createCodingAgent } from "@jai/coding-agent";

const agent = (await createCodingAgent({
  model: "anthropic/claude-sonnet-4-5",
  cwd: process.cwd(),
})).value;

// 按 contentIndex 维护各内容块的增量文本
const blocks = new Map<number, string>();

const unsubscribe = agent.subscribe((event) => {
  switch (event.type) {
    case "agent_start":
      console.log("▶ 运行开始");
      break;
    case "message_update": {
      const ae = event.assistantEvent;
      if (ae.type === "text_delta") {
        blocks.set(ae.contentIndex, (blocks.get(ae.contentIndex) ?? "") + ae.delta);
        process.stdout.write(ae.delta);
      } else if (ae.type === "thinking_delta") {
        // 思考过程：可折叠显示
      } else if (ae.type === "toolcall_end") {
        console.log(`\n[调用] ${ae.toolCall.name}`);
      } else if (ae.type === "done") {
        console.log(`\n[结束] reason=${ae.reason}`);
      }
      break;
    }
    case "tool_execution_start":
      console.log(`[工具开始] ${event.toolName} — ${event.title}`);
      break;
    case "tool_execution_end":
      console.log(`[工具结束] ${event.toolName} isError=${event.isError}`);
      break;
    case "compaction_start":
      console.log(`[压缩] trigger=${event.trigger}`);
      break;
    case "background_agent_end":
      console.log(`[后台结算] ${event.title}: ${event.status}`);
      break;
    case "agent_end":
      console.log("■ 运行结束，共", event.messages.length, "条消息");
      break;
  }
});

await agent.prompt("用 Bash 跑一下测试，顺便看看 git 状态");
await agent.waitForIdle();

unsubscribe();
await agent.close();
```

注意：

- `subscribe` 回调是同步调用的，不要在里面做重型 IO。
- `message_update` 的 `message` 字段是当前 assistant 消息的最新快照，
  也可以直接用它整体重渲染，而不是自己拼 delta。
- `background_agent_end` 可能在其 Operation 结束后才到达——不要假设它
  一定在某次 `prompt` 返回前出现。
