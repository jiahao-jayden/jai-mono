# 示例：恢复会话

```ts
import { createCodingAgent, type CodingSessionStore } from "@jai/coding-agent";

async function resumeSession(sessionId: string, store: CodingSessionStore) {
  const resumed = await createCodingAgent({
    model: "anthropic/claude-sonnet-4-5",
    cwd: "/path/to/project",
    session: { kind: "resume", id: sessionId, store },
    fileCapabilities: {
      homeDirectory: "/Users/you",
      workspaceDirectory: "/path/to/project",
      workspaceTrusted: true,
    },
  });

  if (resumed.isErr()) {
    switch (resumed.error.code) {
      case "coding_sdk.session_not_found":
        console.error(`会话 ${sessionId} 不存在`);
        break;
      case "coding_sdk.session_format_invalid":
        console.error("会话记录不是当前版本格式（需要 version === 1）");
        break;
      default:
        console.error(resumed.error.code, resumed.error.message);
    }
    return;
  }

  const agent = resumed.value;
  // 历史消息已恢复，state.messages 可直接读
  console.log("已恢复，历史消息数:", agent.state.messages.length);

  await agent.prompt("继续之前的工作");
  await agent.close();
}
```

恢复是"按 id 打开 journal"：消息、压缩记录、分支、扩展会话状态都会回来。
`navigate(entryId)` 可在分支间移动（失败 phase 为 `navigation`）。
